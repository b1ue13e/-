## 3. 详细题目与要求

### Task 1: 病态二次型问题上的收敛性定量分析

定义病态二次型函数：
$$f(x) = \frac{1}{2} x^T A x - b^T x, \quad x \in \mathbf{R}^n$$
可以通过改变矩阵 $A$ 的最大与最小特征值比率（条件数 $\kappa = \frac{\lambda_{\max}}{\lambda_{\min}}$），观察 $\kappa = 10, 100, 1000, 10000$ 时算法的表现。

- **理论值比对**：
  - 梯度下降理论单步收缩率 upper bound：$1 - \frac{2}{\kappa + 1}$。
  - 共轭梯度法（CG）误差上限：$2 \left( \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \right)^k$。

#### Python Starter Code 框架：

```python
import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt
import time

# 1. 生成高维病态问题
np.random.seed(42)
n = 50
kappa = 1000 # 条件数 1000

# 构造特征值为 1 到 kappa 等比分布的正定阵
eigenvalues = np.logspace(0, np.log10(kappa), n)
Q, _ = np.linalg.qr(np.random.randn(n, n))
A = Q @ np.diag(eigenvalues) @ Q.T
b = np.random.randn(n)

def f_quad(x):
    return 0.5 * x.T @ A @ x - b.T @ x

def grad_quad(x):
    return A @ x - b

# 2. 手撕带有回溯线搜索的 GD
def gradient_descent(f, grad_f, x0, tol=1e-6, max_iter=2000):
    x = np.array(x0, dtype=float)
    history = [f(x)]
    start_time = time.time()
    
    for k in range(max_iter):
        g = grad_f(x)
        if np.linalg.norm(g) < tol:
            break
        # 单步步长 (在二次型上最佳步长可闭式计算 alpha = g^T g / (g^T A g))
        alpha = np.dot(g, g) / np.dot(g, A @ g)
        x = x - alpha * g
        history.append(f(x))
        
    elapsed = time.time() - start_time
    return x, history, elapsed

# 3. 运行对比（GD vs SciPy CG vs SciPy BFGS）
x0 = np.zeros(n)

# 梯度下降
x_gd, hist_gd, time_gd = gradient_descent(f_quad, grad_quad, x0)

# 共轭梯度 (CG)
hist_cg = []
def cb_cg(xk): hist_cg.append(f_quad(xk))
t0 = time.time()
res_cg = opt.minimize(f_quad, x0, method='CG', jac=grad_quad, callback=cb_cg, options={'gtol': 1e-6})
time_cg = time.time() - t0

# BFGS
hist_bfgs = []
def cb_bfgs(xk): hist_bfgs.append(f_quad(xk))
t0 = time.time()
res_bfgs = opt.minimize(f_quad, x0, method='BFGS', jac=grad_quad, callback=cb_bfgs, options={'gtol': 1e-6})
time_bfgs = time.time() - t0

f_star = f_quad(np.linalg.solve(A, b))

# 4. 可视化收敛曲线 (对数坐标)
plt.figure(figsize=(9, 6))
plt.semilogy([v - f_star for v in hist_gd], label=f'Gradient Descent ({len(hist_gd)} iters, {time_gd:.3f}s)')
plt.semilogy([v - f_star for v in hist_cg], label=f'Conjugate Gradient ({len(hist_cg)} iters, {time_cg:.3f}s)')
plt.semilogy([v - f_star for v in hist_bfgs], label=f'BFGS ({len(hist_bfgs)} iters, {time_bfgs:.3f}s)')

plt.xlabel('Iteration')
plt.ylabel('Suboptimality f(x_k) - f*')
plt.title(f'Convergence Comparison on Ill-conditioned Quadratic ($\kappa={kappa}$)')
plt.grid(True, which="both", ls="--")
plt.legend()
plt.show()

# 检验断言：CG 迭代步数必须远小于 GD
print(f"GD 迭代次数: {len(hist_gd)}, CG 迭代次数: {len(hist_cg)}, BFGS 迭代次数: {len(hist_bfgs)}")
assert len(hist_cg) < len(hist_gd), "共轭梯度法在病态二次型上必须显著优于梯度下降！"
```

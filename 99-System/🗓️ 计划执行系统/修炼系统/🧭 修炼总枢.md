# 🧭 修炼总枢

> 今日所行，皆入命盘；主线一断，心魔即生。这里只结算已有任务，不额外加码。

## 命盘总览

```dataviewjs
const DAY_SECTIONS = [
  { heading: "今日学习", title: "主线天命", xp: 30 },
  { heading: "ChatGPT 出题复盘", title: "试剑台", xp: 10 },
  { heading: "雅思词汇", title: "外门功课", xp: 10 },
  { heading: "训练计划", title: "炼体", xp: 30 },
  { heading: "饮食与身体打卡", title: "辟谷", xp: 5 },
  { heading: "当日收束", title: "夜观命盘", xp: 5 },
];
const REALMS = ["炼气", "筑基", "金丹", "元婴", "化神", "炼虚", "合体", "大乘", "渡劫"];
const LAYERS = ["一", "二", "三", "四", "五", "六", "七", "八", "九"];
const FINAL_TITLE = "万法归一·诸天道祖";
const today = dv.luxon.DateTime.now().startOf("day");

const toDate = value => {
  if (!value) return null;
  if (value.toISODate) return value.startOf ? value.startOf("day") : value;
  const parsed = dv.date(String(value));
  return parsed ? parsed.startOf("day") : null;
};
const pct = (num, den) => den > 0 ? `${Math.round(num / den * 100)}%` : "-";
const clamp = (value, min, max) => Math.max(min, Math.min(max, value));
const ignoredPagePaths = new Set(["AGENTS.md", "README.md"]);
const ignoredDirPrefixes = [
  ".obsidian/",
  ".trash/",
  ".tmp.drivedownload/",
  ".tmp.driveupload/",
  "tmp/",
  "tools/",
  "99-System/🗓️ 计划执行系统/修炼系统/"
];
const isManagedPage = p =>
  p?.file?.path &&
  !p.template &&
  !ignoredPagePaths.has(p.file.path) &&
  !ignoredDirPrefixes.some(prefix => p.file.path.startsWith(prefix));

function sectionLines(text, heading) {
  const lines = text.split(/\r?\n/);
  const out = [];
  let inside = false;
  let level = 0;
  for (const line of lines) {
    const match = line.match(/^(#{2,6})\s+(.+?)\s*$/);
    if (match) {
      const currentLevel = match[1].length;
      const title = match[2].trim();
      if (inside && currentLevel <= level) break;
      if (!inside && currentLevel === 2 && title === heading) {
        inside = true;
        level = currentLevel;
        continue;
      }
    }
    if (inside) out.push(line);
  }
  return out;
}
function sectionStats(text, heading) {
  const tasks = sectionLines(text, heading).filter(line => /^-\s+\[[ xX]\]/.test(line));
  return {
    total: tasks.length,
    done: tasks.filter(line => /^-\s+\[[xX]\]/.test(line)).length,
  };
}
function dayStats(text) {
  const sections = DAY_SECTIONS.map(rule => {
    const stat = sectionStats(text, rule.heading);
    return {
      ...rule,
      total: stat.total,
      done: stat.done,
      xpEarned: stat.done * rule.xp,
      xpMax: stat.total * rule.xp,
    };
  });
  const main = sections.find(s => s.heading === "今日学习");
  const sword = sections.find(s => s.heading === "ChatGPT 出题复盘");
  const close = sections.find(s => s.heading === "当日收束");
  const perfect =
    main?.total > 0 && main.done === main.total &&
    sword?.total > 0 && sword.done === sword.total &&
    close?.total > 0 && close.done === close.total;
  let demonDelta = 0;
  if (main?.total > 0) {
    if (main.done <= 1) demonDelta += 2;
    else if (main.done === 2) demonDelta += 1;
  }
  if (perfect) demonDelta -= 1;
  return {
    sections,
    main,
    perfect,
    demonDelta,
    xpEarned: sections.reduce((sum, s) => sum + s.xpEarned, 0),
    xpMax: sections.reduce((sum, s) => sum + s.xpMax, 0),
    taskDone: sections.reduce((sum, s) => sum + s.done, 0),
    taskTotal: sections.reduce((sum, s) => sum + s.total, 0),
  };
}
function parseProgress(value) {
  if (value === null || value === undefined) return 0;
  if (typeof value === "number") return value <= 1 ? value * 100 : value;
  const match = String(value).match(/(\d+(?:\.\d+)?)/);
  return match ? Number(match[1]) : 0;
}
function pageXp(page) {
  const type = page.type;
  const status = page.status;
  if (type === "study") return { earned: status === "completed" ? 10 : status === "ongoing" ? 3 : 0, max: 10 };
  if (type === "stage") return { earned: status === "completed" ? 80 : ["active", "ongoing"].includes(status) ? 20 : 0, max: 80 };
  if (type === "paper") return { earned: status === "finished" ? 120 : status === "reading" ? 40 : 0, max: 120 };
  if (type === "book") return { earned: status === "finished" ? 200 : status === "reading" ? 60 : 0, max: 200 };
  if (type === "project") {
    const progress = clamp(parseProgress(page.progress), 0, 100);
    return { earned: status === "completed" ? 300 : Math.round(300 * progress / 100), max: 300 };
  }
  return { earned: 0, max: 0 };
}
function layerFromRatio(earned, max) {
  if (max <= 0 || earned <= 0) return 0;
  if (earned >= max) return 81;
  return clamp(Math.floor(earned / max * 81) + 1, 1, 81);
}
function realmName(layer) {
  if (layer >= 81) return "渡劫九层";
  if (layer <= 0) return "凡人未入道";
  const index = layer - 1;
  return `${REALMS[Math.floor(index / 9)]}${LAYERS[index % 9]}层`;
}
function applyDemon(layer, demon) {
  if (layer <= 0) return 0;
  const penalty = Math.floor(Math.max(0, demon) / 2);
  const realmBase = Math.floor((layer - 1) / 9) * 9 + 1;
  return Math.max(realmBase, layer - penalty);
}

const dayPages = dv.pages('"99-System/🗓️ 计划执行系统/日计划"')
  .where(p => p.start && !p.template)
  .array()
  .sort((a, b) => toDate(a.start).ts - toDate(b.start).ts);

let dailyXp = 0;
let dailyMax = 0;
let demonRaw = 0;
let todayRecord = null;
const recentRecords = [];
for (const page of dayPages) {
  const date = toDate(page.start);
  if (!date) continue;
  const text = await dv.io.load(page.file.path);
  const stat = dayStats(text);
  dailyXp += stat.xpEarned;
  dailyMax += stat.xpMax;
  if (date < today) demonRaw += stat.demonDelta;
  if (date.toISODate() === today.toISODate()) todayRecord = { page, date, stat };
  if (date <= today && date >= today.minus({ days: 6 })) recentRecords.push({ page, date, stat });
}

let longXp = 0;
let longMax = 0;
const typeBuckets = {};
for (const page of dv.pages().where(isManagedPage).array()) {
  const score = pageXp(page);
  if (score.max <= 0) continue;
  longXp += score.earned;
  longMax += score.max;
  const key = page.type;
  if (!typeBuckets[key]) typeBuckets[key] = { count: 0, earned: 0, max: 0, completed: 0 };
  typeBuckets[key].count += 1;
  typeBuckets[key].earned += score.earned;
  typeBuckets[key].max += score.max;
  if (["completed", "finished"].includes(page.status)) typeBuckets[key].completed += 1;
}

const totalXp = dailyXp + longXp;
const totalMax = dailyMax + longMax;
const demon = Math.max(0, demonRaw);
const highestLayer = layerFromRatio(totalXp, totalMax);
const currentLayer = applyDemon(highestLayer, demon);
const finalReached = totalMax > 0 && totalXp >= totalMax && demon === 0;
const nextLayerXp = highestLayer >= 81 ? 0 : Math.max(0, Math.ceil((highestLayer / 81) * totalMax - totalXp));

dv.table(
  ["命盘", "当前"],
  [
    ["当前境界", finalReached ? FINAL_TITLE : realmName(currentLayer)],
    ["历史最高境界", finalReached ? FINAL_TITLE : realmName(highestLayer)],
    ["总修为", `${totalXp} / ${totalMax} (${pct(totalXp, totalMax)})`],
    ["下一突破", highestLayer >= 81 ? "已达渡劫九层" : `还需 ${nextLayerXp} 修为`],
    ["心魔值", demon === 0 ? "0，心境澄明" : `${demon}，当前压低 ${Math.floor(demon / 2)} 小层`],
    ["今日完成率", todayRecord ? pct(todayRecord.stat.taskDone, todayRecord.stat.taskTotal) : "今日无日计划"],
  ]
);
```
